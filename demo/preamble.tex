%% FIX: Load the renamed package by package name to match \ProvidesPackage.
\usepackage{lanepaper}

%% Issue #48 (ADR-0003): the document owns these document-level packages.
%% Lanepaper configures only Hyperref link styling and longtable caption width
%% if present. Order matters between hyperref and cleveref.
%% The bibliography is the document's since #48. These are the options the
%% package used to apply on its own; change them freely, that is the point.
%% FIX (#86): The document loads csquotes before biblatex now that Lanepaper
%% no longer supplies quotation-package dependencies.
%% FIX (#102): Keep only document-owned packages used by this curated showcase.
\usepackage{csquotes}
\usepackage[
  backend=biber,
  style=authoryear,
  natbib=true,
  sorting=nyt,
  maxcitenames=2,
  maxbibnames=99,
  giveninits=true,
  uniquename=init,
  doi=true,
  url=true,
  isbn=false
]{biblatex}
\usepackage[english]{babel}  % language and hyphenation patterns
\usepackage{hyperref}    % PDF links and metadata; load late, before cleveref
\usepackage{cleveref}    % cross-references; must follow hyperref

% Load the bibliography database
\addbibresource{references.bib}
